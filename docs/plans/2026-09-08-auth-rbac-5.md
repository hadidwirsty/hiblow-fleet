```typescript
"use client"

import { useState } from "react"
import { useRouter } from "next/navigation"
import { z } from "zod"
import { RiLockLine, RiMailLine, RiTruckLine } from "@remixicon/react"

import { Button } from "@/components/ui/button"
import { Card, CardContent, CardDescription, CardHeader, CardTitle } from "@/components/ui/card"
import { Input } from "@/components/ui/input"
import { Label } from "@/components/ui/label"
import { signIn } from "@/lib/auth-client"

const loginSchema = z.object({
  email: z.string().email("Email tidak valid"),
  password: z.string().min(6, "Password minimal 6 karakter"),
})

export function LoginForm() {
  const router = useRouter()
  const [email, setEmail] = useState("")
  const [password, setPassword] = useState("")
  const [error, setError] = useState<string | null>(null)
  const [isPending, setIsPending] = useState(false)

  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault()
    setError(null)

    const validated = loginSchema.safeParse({ email, password })
    if (!validated.success) {
      setError(validated.error.errors[0].message)
      return
    }

    setIsPending(true)
    try {
      const result = await signIn.email({
        email: validated.data.email,
        password: validated.data.password,
      })
      if (result.error) {
        setError("Email atau password salah.")
        return
      }
      // Redirect berdasarkan role (middleware akan handle jika salah arah)
      const role = result.data?.user?.role
      router.replace(role === "admin" ? "/dashboard" : "/profit-sharing")
    } catch {
      setError("Terjadi kesalahan. Coba lagi.")
    } finally {
      setIsPending(false)
    }
  }

  return (
    <Card className="w-full max-w-sm shadow-lg">
      <CardHeader className="text-center pb-4">
        <div className="mx-auto mb-3 flex size-12 items-center justify-center rounded-xl bg-primary text-primary-foreground shadow-sm">
          <RiTruckLine className="size-6" />
        </div>
        <CardTitle className="text-xl font-bold tracking-tight">HW Trans Fleet</CardTitle>
        <CardDescription className="text-xs">
          Masuk ke sistem manajemen armada & keuangan
        </CardDescription>
      </CardHeader>
      <CardContent>
        <form onSubmit={handleSubmit} className="space-y-4">
          <div className="space-y-1.5">
            <Label htmlFor="email" className="text-xs font-medium">Email</Label>
            <div className="relative">
              <RiMailLine className="absolute left-2.5 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" />
              <Input
                id="email"
                type="email"
                placeholder="hafidz@hiblow.fleet"
                value={email}
                onChange={(e) => setEmail(e.target.value)}
                className="pl-8 text-sm"
                required
              />
            </div>
          </div>
          <div className="space-y-1.5">
            <Label htmlFor="password" className="text-xs font-medium">Password</Label>
            <div className="relative">
              <RiLockLine className="absolute left-2.5 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" />
              <Input
                id="password"
                type="password"
                placeholder="••••••••"
                value={password}
                onChange={(e) => setPassword(e.target.value)}
                className="pl-8 text-sm"
                required
              />
            </div>
          </div>
          {error && (
            <p className="rounded-md bg-destructive/10 px-3 py-2 text-xs text-destructive">
              {error}
            </p>
          )}
          <Button type="submit" className="w-full" disabled={isPending}>
            {isPending ? "Memproses..." : "Masuk"}
          </Button>
        </form>
      </CardContent>
    </Card>
  )
}
```

### Task 2: Middleware Proteksi Route

**Files:**
- Create: `middleware.ts`
- Test: `domain/__tests__/auth-rbac.test.ts`

**Requirements:**

**Acceptance Criteria**
1. Semua route di bawah `/(dashboard)` membutuhkan sesi aktif; tanpa sesi redirect ke `/login`.
2. Route `/login` yang diakses oleh user yang sudah login → redirect ke tujuan berdasarkan role.
3. Route `/dashboard`, `/trips`, `/expenses`, `/rates` hanya boleh diakses oleh `admin`.
4. Route `/profit-sharing` boleh diakses oleh `admin` dan `partner`.
5. `partner` yang mencoba akses `/dashboard`, `/trips`, `/expenses`, `/rates` → redirect ke `/profit-sharing`.

**Functional Requirements**
1. Middleware menggunakan `auth.api.getSession()` dengan `fromNodeHeaders()` dari Better Auth.
2. Gunakan `NextResponse.redirect()` untuk semua kasus proteksi.
3. Matcher mencakup semua route kecuali: `/_next/`, `/api/auth/`, `/favicon.ico`, dan file statis.

**Non-Functional Requirements**
- Middleware dieksekusi di Edge Runtime; gunakan API yang kompatibel.
- Tidak boleh mengekspos detail peran di URL redirect (cukup `/profit-sharing` atau `/login`).

**Test Coverage**
- [Unit] `determineRedirectPath(role, pathname)` — pure function helper, testable tanpa Next.js runtime.

```typescript
// Pure helper function (akan diekstrak dari middleware)
function determineRedirectPath(
  role: string | null | undefined,
  pathname: string,
  isAuthenticated: boolean
): string | null {
  if (!isAuthenticated) return "/login"
  if (pathname === "/login" || pathname === "/") {
    return role === "admin" ? "/dashboard" : "/profit-sharing"
  }
  const adminOnlyRoutes = ["/dashboard", "/trips", "/expenses", "/rates"]
  const isAdminOnly = adminOnlyRoutes.some((r) => pathname.startsWith(r))
  if (isAdminOnly && role !== "admin") return "/profit-sharing"
  return null // no redirect
}

describe("determineRedirectPath", () => {
  it("harus redirect ke /login jika tidak terautentikasi", () => {
    expect(determineRedirectPath(null, "/dashboard", false)).toBe("/login")
```

**`middleware.ts`:**
```typescript
import { NextResponse } from "next/server"

import { auth } from "@/lib/auth"

import type { NextRequest } from "next/server"

const ADMIN_ONLY_ROUTES = ["/dashboard", "/trips", "/expenses", "/rates"]

export async function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl

  const session = await auth.api.getSession({
    headers: request.headers,
  })

  const isAuthenticated = !!session?.user
  const role = session?.user?.role ?? null

  // Unauthenticated user → redirect ke /login
  if (!isAuthenticated) {
    const loginUrl = new URL("/login", request.url)
    return NextResponse.redirect(loginUrl)
  }

  // Authenticated user di halaman /login → redirect ke tujuan berdasarkan role
  if (pathname === "/login") {
    const target = role === "admin" ? "/dashboard" : "/profit-sharing"
    return NextResponse.redirect(new URL(target, request.url))
  }

  // Partner mencoba akses route admin-only → redirect ke /profit-sharing
  const isAdminOnly = ADMIN_ONLY_ROUTES.some((r) => pathname.startsWith(r))
  if (isAdminOnly && role !== "admin") {
    return NextResponse.redirect(new URL("/profit-sharing", request.url))
  }

  return NextResponse.next()
}

export const config = {
  matcher: [
    "/((?!_next/static|_next/image|favicon.ico|api/auth).*)",
  ],
}
```
